```python
"""Board 领域对象：持有 Cell 网格并回答邻居/边界查询。"""

from dataclasses import dataclass, field
from typing import Iterator

from .cell import Cell
from .coordinate import Coordinate


@dataclass
class Board:
    """rows×cols 的格子网格。"""

    rows: int
    cols: int
    _cells: list[list[Cell]] = field(init=False, repr=False)

    def __post_init__(self) -> None:
        """初始化全部格子。"""
        self._cells = [[Cell() for _ in range(self.cols)] for _ in range(self.rows)]

    def in_bounds(self, coord: Coordinate) -> bool:
        """判断坐标是否在棋盘内。"""
        return 0 <= coord.row < self.rows and 0 <= coord.col < self.cols

    def cell(self, coord: Coordinate) -> Cell:
        """取指定坐标的格子对象。"""
        return self._cells[coord.row][coord.col]

    def all_coords(self) -> Iterator[Coordinate]:
        """遍历棋盘内全部坐标。"""
        for row in range(self.rows):
            for col in range(self.cols):
                yield Coordinate(row, col)

    def neighbors8(self, coord: Coordinate) -> list[Coordinate]:
        """返回 3×3 内 8 个邻居（跳过越界与自身）。"""
        result: list[Coordinate] = []
        for dr in (-1, 0, 1):
            for dc in (-1, 0, 1):
                if dr == 0 and dc == 0:
                    continue
                nxt = Coordinate(coord.row + dr, coord.col + dc)
                if self.in_bounds(nxt):
                    result.append(nxt)
        return result
```

```python
"""难度数据类与 JSON 加载。"""

import json
from dataclasses import dataclass
from pathlib import Path


@dataclass(frozen=True)
class Difficulty:
    """一种难度配置。"""

    rows: int
    cols: int
    mines: int


@dataclass(frozen=True)
class DifficultyConfig:
    """全部难度预设与自定义范围。"""

    presets: dict[str, Difficulty]
    min_rows: int
    max_rows: int
    min_cols: int
    max_cols: int
    min_mines: int

    def preset(self, name: str) -> Difficulty:
        """按名字取预设难度。"""
        return self.presets[name]


def load_difficulties(path: Path) -> DifficultyConfig:
    """从 JSON 解析难度配置。"""
    data = json.loads(path.read_text(encoding="utf-8"))
    presets = {
        name: Difficulty(rows=v["rows"], cols=v["cols"], mines=v["mines"])
        for name, v in data.items()
        if name != "custom_limits"
    }
    limits = data["custom_limits"]
    return DifficultyConfig(
        presets=presets,
        min_rows=limits["min_rows"],
        max_rows=limits["max_rows"],
        min_cols=limits["min_cols"],
        max_cols=limits["max_cols"],
        min_mines=limits["min_mines"],
    )


def validate_custom(rows: int, cols: int, mines: int, cfg: DifficultyConfig) -> None:
    """校验自定义难度，非法抛 ValueError（开发文档 §4.4）。"""
    if not (cfg.min_rows <= rows <= cfg.max_rows):
        raise ValueError(f"行数需在 {cfg.min_rows}~{cfg.max_rows}")
    if not (cfg.min_cols <= cols <= cfg.max_cols):
        raise ValueError(f"列数需在 {cfg.min_cols}~{cfg.max_cols}")
    if not (cfg.min_mines <= mines <= min(999, rows * cols - 9)):
        raise ValueError("雷数不合法")
```

## 4. Core：布雷（开发文档 §6.3）

`core/generation/mine_generator.py`：

```python
"""点后布雷。"""

import random

from ..model.board import Board
from ..model.coordinate import Coordinate


def _safe_zone(click: Coordinate, mode: str, rows: int, cols: int) -> set[Coordinate]:
    """计算禁止布雷区：cell=仅首格，zone3x3=首格 3×3。"""
    if mode == "cell":
        return {click}
    zone: set[Coordinate] = set()
    for dr in (-1, 0, 1):
        for dc in (-1, 0, 1):
            c = Coordinate(click.row + dr, click.col + dc)
            if 0 <= c.row < rows and 0 <= c.col < cols:
                zone.add(c)
    return zone


def place_mines(
    board: Board,
    mine_count: int,
    first_click: Coordinate,
    safe_mode: str,
    rng: random.Random,
) -> None:
    """在安全区之外的格子中随机放置 mine_count 颗雷。"""
    forbidden = _safe_zone(first_click, safe_mode, board.rows, board.cols)
    candidates = [c for c in board.all_coords() if c not in forbidden]
    for c in rng.sample(candidates, mine_count):
        board.cell(c).is_mine = True
```

> 布雷完成后必须调用 §5 的 `calculate_adjacent_mines(board)` 预计算邻域雷数，否则翻开阶段读到的 `adjacent_mines` 全是初始值。

## 5. Core：翻开与洪水填充（开发文档 §6.4）

`core/rules/reveal.py`：

```python
"""翻开与 BFS 洪水填充（邻域雷数在布雷后预计算）。"""

from collections import deque

from ..enums import Mark
from ..model.board import Board
from ..model.coordinate import Coordinate


def count_adjacent_mines(board: Board, coord: Coordinate) -> int:
    """统计 3×3 邻域内的雷数。"""
    return sum(1 for n in board.neighbors8(coord) if board.cell(n).is_mine)


def calculate_adjacent_mines(board: Board) -> None:
    """布雷后调用：遍历棋盘，为每个格子预计算邻域雷数。"""
    for coord in board.all_coords():
        board.cell(coord).adjacent_mines = count_adjacent_mines(board, coord)


def reveal_cell(board: Board, coord: Coordinate) -> bool:
    """翻开单格；返回是否确实翻开（雷/旗/已翻开/越界返回 False）。"""
    if not board.in_bounds(coord):
        return False
    cell = board.cell(coord)
    if cell.is_revealed or cell.is_mine or cell.mark is Mark.FLAG:
        return False
    cell.is_revealed = True
    return True


def flood_fill(board: Board, start: Coordinate) -> list[Coordinate]:
    """从 start 展开空白区（读预计算的 adjacent_mines），返回本次被翻开的全部格子。"""
    revealed: list[Coordinate] = []
    queue: deque[Coordinate] = deque()
    queue.append(start)
    while queue:
        cur = queue.popleft()
        if not reveal_cell(board, cur):
            continue
        revealed.append(cur)
        if board.cell(cur).adjacent_mines == 0:
            queue.extend(board.neighbors8(cur))
    return revealed
```

> `calculate_adjacent_mines` 只在布雷后调用一次（见 §11 首次点击），翻开/连开阶段只读取预计算结果，不再逐格统计。

```python
"""右键旗/问号循环。"""

from ..enums import Mark
from ..model.cell import Cell


def cycle_mark(cell: Cell, marks_enabled: bool) -> int:
    """循环标记；返回剩余雷数的变化量（标旗 -1，去旗 +1，问号 0）。"""
    if cell.is_revealed:
        return 0
    if cell.mark is Mark.NONE:
        cell.mark = Mark.FLAG
        return -1
    if cell.mark is Mark.FLAG:
        cell.mark = Mark.QUESTION if marks_enabled else Mark.NONE
        return 1
    cell.mark = Mark.NONE
    return 0
```

## 7. Core：连开 chord（开发文档 §6.5）

`core/rules/chord.py`：

```python
"""中键/双键连开。"""

from ..enums import Mark
from ..model.board import Board
from ..model.coordinate import Coordinate
from .reveal import flood_fill


def chord(board: Board, coord: Coordinate) -> tuple[list[Coordinate], bool]:
    """连开周围格子；返回 (本次翻开的格子, 是否踩雷)。"""
    cell = board.cell(coord)
    if not cell.is_revealed:
        return [], False
    flags = sum(1 for n in board.neighbors8(coord) if board.cell(n).mark is Mark.FLAG)
    if cell.adjacent_mines != flags:
        return [], False
    opened: list[Coordinate] = []
    for n in board.neighbors8(coord):
        neighbor = board.cell(n)
        if neighbor.mark is Mark.FLAG:
            continue
        if neighbor.is_mine:
            return opened, True
        opened.extend(flood_fill(board, n))
    return opened, False
```

> 连开踩雷时返回**布尔值**；`opened` 只包含真正被翻开的格子，雷不进入 `opened`（雷不会被翻开）。
> UI 终局重绘时以 Game 记录的触发坐标为中心：`neighbors8(触发坐标)` 中「是雷且未标旗」的格子整片高亮，
> 标错的旗画精灵 11（错旗），其余未标旗的雷画精灵 10（终局雷），具体见开发文档 §4.3。

```python
"""胜负判定。"""

from ..model.board import Board


def is_won(board: Board, mine_count: int) -> bool:
    """所有非雷格全部翻开即胜（雷是否标旗无关）。"""
    target = board.rows * board.cols - mine_count
    revealed_safe = sum(
        1
        for coord in board.all_coords()
        if board.cell(coord).is_revealed and not board.cell(coord).is_mine
    )
    return revealed_safe == target
```

## 9. Core：Clock（开发文档 §6.7）

`core/clock.py`：

```python
"""计时抽象。Core 不依赖 QTimer。"""

from typing import Protocol


class Clock(Protocol):
    """计时接口。"""

    def start(self) -> None: ...
    def stop(self) -> None: ...
    def pause(self) -> None: ...
    def resume(self) -> None: ...

    @property
    def elapsed(self) -> int: ...


class FakeClock:
    """手动推进的时钟，供测试使用。"""

    def __init__(self) -> None:
        self._running = False
        self._paused = False
        self._elapsed = 0

    def start(self) -> None:
        self._running = True

    def stop(self) -> None:
        self._running = False

    def pause(self) -> None:
        self._paused = True

    def resume(self) -> None:
        self._paused = False

    def tick(self, seconds: int = 1) -> None:
        """推进 elapsed（仅在运行且未暂停时）。"""
        if self._running and not self._paused:
            self._elapsed = min(999, self._elapsed + seconds)

    @property
    def elapsed(self) -> int:
        return self._elapsed
```

## 11. Core：Game（开发文档 §5.4 / §6.8）

`core/game.py`：

```python
"""Game 聚合根：持有状态、分发命令、发布事件。"""

import random
from dataclasses import dataclass
from typing import Callable

from .clock import Clock, FakeClock
from .command import ChangeDifficulty, ChordCell, Command, RestartGame, RevealCell, ToggleMark
from .enums import FaceState, GameStatus, Mark
from .event import (
    CellMarked,
    CellRevealed,
    CellsRevealed,
    Event,
    FaceChanged,
    FlagsChanged,
    GameLost,
    GameStateChanged,
    GameWon,
    TimerTicked,
)
from .generation.mine_generator import place_mines
from .model.board import Board
from .model.coordinate import Coordinate
from .rules import chord, marking, reveal, victory


@dataclass
class GameState:
    """游戏状态快照（开发文档 §5.4）。"""

    status: GameStatus
    difficulty: str
    rows: int
    cols: int
    mine_count: int
    remaining_mines: int
    elapsed_seconds: int
    face_state: FaceState


class Game:
    """扫雷逻辑核心，不含任何 Qt。"""

    def __init__(self, clock: Clock | None = None) -> None:
        self._clock: Clock = clock or FakeClock()
        self._listeners: list[Callable[[Event], None]] = []
        self._board = Board(9, 9)
        self._rng = random.Random()
        self._first_click: Coordinate | None = None
        self.safe_mode = "zone3x3"
        self.marks_enabled = True
        self.state = GameState(GameStatus.READY, "beginner", 9, 9, 10, 10, 0, FaceState.NORMAL)

    @property
    def board(self) -> Board:
        return self._board

    def subscribe(self, listener: Callable[[Event], None]) -> None:
        """注册事件监听器。"""
        self._listeners.append(listener)

    def _emit(self, event: Event) -> None:
        for listener in self._listeners:
            listener(event)

    def new_game(self, difficulty: str, rows: int, cols: int, mines: int) -> None:
        """开新局：清盘、重置状态，布雷推迟到首次点击。"""
        self._clock.stop()
        self._board = Board(rows, cols)
        self._first_click = None
        self.state = GameState(
            GameStatus.READY, difficulty, rows, cols, mines, mines, 0, FaceState.NORMAL
        )
        self._emit(GameStateChanged(self.state.status))
        self._emit(FlagsChanged(mines))
        self._emit(FaceChanged(FaceState.NORMAL))

    def dispatch(self, command: Command) -> None:
        """分发命令（终局后仅 Restart 有效，见开发文档 §6.8）。"""
        if isinstance(command, RestartGame):
            self.new_game(
                self.state.difficulty, self.state.rows, self.state.cols, self.state.mine_count
            )
            return
        if self.state.status in (GameStatus.WON, GameStatus.LOST):
            return
        if isinstance(command, ChangeDifficulty):
            self.new_game(command.name, self.state.rows, self.state.cols, self.state.mine_count)
        elif isinstance(command, RevealCell):
            self._reveal(command.coord)
        elif isinstance(command, ToggleMark):
            self._mark(command.coord)
        elif isinstance(command, ChordCell):
            self._chord(command.coord)

    def tick(self) -> None:
        """每秒被 UI 的 QTimer 调用，发布计时事件。"""
        if self.state.status is GameStatus.PLAYING and self._clock.elapsed >= 999:
            self._clock.stop()
        self._emit(TimerTicked(self._clock.elapsed))

    def _reveal(self, coord: Coordinate) -> None:
        if not self._board.in_bounds(coord):
            return
        if self._first_click is None:
            self._first_click = coord
            place_mines(self._board, self.state.mine_count, coord, self.safe_mode, self._rng)
            reveal.calculate_adjacent_mines(self._board)
            self._clock.start()
            self.state.status = GameStatus.PLAYING
            self._emit(GameStateChanged(GameStatus.PLAYING))
        cell = self._board.cell(coord)
        if cell.is_mine:
            self._last_trigger = coord  # 直接踩雷：记录触发坐标供 UI 终局高亮
            self._end(GameStatus.LOST)
            return
        opened = reveal.flood_fill(self._board, coord)
        if opened:
            self._emit(
                CellsRevealed(tuple((c, self._board.cell(c).adjacent_mines) for c in opened))
            )
        if victory.is_won(self._board, self.state.mine_count):
            self._end(GameStatus.WON)

    def _mark(self, coord: Coordinate) -> None:
        if not self._board.in_bounds(coord):
            return
        cell = self._board.cell(coord)
        delta = marking.cycle_mark(cell, self.marks_enabled)
        self.state.remaining_mines += delta
        self._emit(CellMarked(coord, cell.mark))
        self._emit(FlagsChanged(self.state.remaining_mines))

    def _chord(self, coord: Coordinate) -> None:
        if not self._board.in_bounds(coord):
            return
        opened, hit_mine = chord.chord(self._board, coord)
        if opened:
            self._emit(
                CellsRevealed(tuple((c, self._board.cell(c).adjacent_mines) for c in opened))
            )
        if hit_mine:
            self._last_trigger = coord  # 供 UI 终局重绘：以触发坐标为中心计算未标旗雷区高亮
            self._end(GameStatus.LOST)
        elif victory.is_won(self._board, self.state.mine_count):
            self._end(GameStatus.WON)

    def _end(self, status: GameStatus) -> None:
        self._clock.stop()
        self.state.status = status
        self.state.face_state = FaceState.WINNER if status is GameStatus.WON else FaceState.LOST
        self.state.elapsed_seconds = self._clock.elapsed
        self._emit(FaceChanged(self.state.face_state))
        self._emit(GameStateChanged(status))
        self._emit(GameWon() if status is GameStatus.WON else GameLost())
```

```python
"""QTimer 驱动的时钟。"""

from PySide6.QtCore import QTimer

from minesweeper_xp.core.clock import Clock


class QtClock(Clock):
    """每 1000ms 递增 elapsed 的时钟。"""

    def __init__(self) -> None:
        self._elapsed = 0
        self._timer = QTimer()
        self._timer.setInterval(1000)
        self._timer.timeout.connect(self._on_timeout)

    def _on_timeout(self) -> None:
        self._elapsed = min(999, self._elapsed + 1)

    def start(self) -> None:
        self._timer.start()

    def stop(self) -> None:
        self._timer.stop()

    def pause(self) -> None:
        self._timer.stop()

    def resume(self) -> None:
        self._timer.start()

    @property
    def elapsed(self) -> int:
        return self._elapsed
```

```python
"""棋盘控件（Vertical Slice 最小实现）。"""

from PySide6.QtCore import Qt, QTimer, Signal
from PySide6.QtGui import QMouseEvent, QPainter
from PySide6.QtWidgets import QWidget

from minesweeper_xp.core.command import ChordCell, RevealCell, ToggleMark
from minesweeper_xp.core.enums import GameStatus, Mark
from minesweeper_xp.core.game import Game
from minesweeper_xp.core.model.coordinate import Coordinate


class BoardWidget(QWidget):
    """把鼠标事件转成 Command 交给 Game，再根据事件重绘。"""

    game_changed = Signal()

    def __init__(self, game: Game) -> None:
        super().__init__()
        self._game = game
        self._game.subscribe(lambda _event: self.game_changed.emit())
        self.game_changed.connect(self.update)
        self.setMouseTracking(True)
        self._tick_timer = QTimer(self)
        self._tick_timer.setInterval(1000)
        self._tick_timer.timeout.connect(self._game.tick)
        self._tick_timer.start()

    def cell_size(self) -> int:
        return 16

    def _coord_at(self, pos) -> Coordinate:
        return Coordinate(pos.y() // self.cell_size(), pos.x() // self.cell_size())

    def mousePressEvent(self, event: QMouseEvent) -> None:
        coord = self._coord_at(event.position())
        if not self._game.board.in_bounds(coord):
            return
        if event.button() == Qt.MouseButton.LeftButton:
            self._game.dispatch(RevealCell(coord))
        elif event.button() == Qt.MouseButton.RightButton:
            self._game.dispatch(ToggleMark(coord))
        elif event.button() == Qt.MouseButton.MiddleButton:
            self._game.dispatch(ChordCell(coord))

    def paintEvent(self, _event) -> None:
        painter = QPainter(self)
        s = self.cell_size()
        for coord in self._game.board.all_coords():
            cell = self._game.board.cell(coord)
            x, y = coord.col * s, coord.row * s
            if cell.is_revealed:
                painter.fillRect(x, y, s, s, Qt.GlobalColor.lightGray)
                if cell.is_mine:
                    painter.fillRect(x, y, s, s, Qt.GlobalColor.black)
                elif cell.adjacent_mines:
                    painter.drawText(
                        x, y, s, s, Qt.AlignmentFlag.AlignCenter, str(cell.adjacent_mines)
                    )
            else:
                painter.fillRect(x, y, s, s, Qt.GlobalColor.gray)
                if cell.mark is Mark.FLAG:
                    painter.drawText(x, y, s, s, Qt.AlignmentFlag.AlignCenter, "F")
                elif cell.mark is Mark.QUESTION:
                    painter.drawText(x, y, s, s, Qt.AlignmentFlag.AlignCenter, "?")
```

```python
"""Board 领域对象：持有 Cell 网格并回答邻居/边界查询。"""

from dataclasses import dataclass, field
from typing import Iterator

from .cell import Cell
from .coordinate import Coordinate


@dataclass
class Board:
    """rows×cols 的格子网格。"""

    rows: int
    cols: int
    _cells: list[list[Cell]] = field(init=False, repr=False)

    def __post_init__(self) -> None:
        """初始化全部格子。"""
        self._cells = [[Cell() for _ in range(self.cols)] for _ in range(self.rows)]

    def in_bounds(self, coord: Coordinate) -> bool:
        """判断坐标是否在棋盘内。"""
        return 0 <= coord.row < self.rows and 0 <= coord.col < self.cols

    def cell(self, coord: Coordinate) -> Cell:
        """取指定坐标的格子对象。"""
        return self._cells[coord.row][coord.col]

    def all_coords(self) -> Iterator[Coordinate]:
        """遍历棋盘内全部坐标。"""
        for row in range(self.rows):
            for col in range(self.cols):
                yield Coordinate(row, col)

    def neighbors8(self, coord: Coordinate) -> list[Coordinate]:
        """返回 3×3 内 8 个邻居（跳过越界与自身）。"""
        result: list[Coordinate] = []
        for dr in (-1, 0, 1):
            for dc in (-1, 0, 1):
                if dr == 0 and dc == 0:
                    continue
                nxt = Coordinate(coord.row + dr, coord.col + dc)
                if self.in_bounds(nxt):
                    result.append(nxt)
        return result
```

```python
"""难度数据类与 JSON 加载。"""

import json
from dataclasses import dataclass
from pathlib import Path


@dataclass(frozen=True)
class Difficulty:
    """一种难度配置。"""

    rows: int
    cols: int
    mines: int


@dataclass(frozen=True)
class DifficultyConfig:
    """全部难度预设与自定义范围。"""

    presets: dict[str, Difficulty]
    min_rows: int
    max_rows: int
    min_cols: int
    max_cols: int
    min_mines: int

    def preset(self, name: str) -> Difficulty:
        """按名字取预设难度。"""
        return self.presets[name]


def load_difficulties(path: Path) -> DifficultyConfig:
    """从 JSON 解析难度配置。"""
    data = json.loads(path.read_text(encoding="utf-8"))
    presets = {
        name: Difficulty(rows=v["rows"], cols=v["cols"], mines=v["mines"])
        for name, v in data.items()
        if name != "custom_limits"
    }
    limits = data["custom_limits"]
    return DifficultyConfig(
        presets=presets,
        min_rows=limits["min_rows"],
        max_rows=limits["max_rows"],
        min_cols=limits["min_cols"],
        max_cols=limits["max_cols"],
        min_mines=limits["min_mines"],
    )


def validate_custom(rows: int, cols: int, mines: int, cfg: DifficultyConfig) -> None:
    """校验自定义难度，非法抛 ValueError（开发文档 §4.4）。"""
    if not (cfg.min_rows <= rows <= cfg.max_rows):
        raise ValueError(f"行数需在 {cfg.min_rows}~{cfg.max_rows}")
    if not (cfg.min_cols <= cols <= cfg.max_cols):
        raise ValueError(f"列数需在 {cfg.min_cols}~{cfg.max_cols}")
    if not (cfg.min_mines <= mines <= min(999, rows * cols - 9)):
        raise ValueError("雷数不合法")
```

## 4. Core：布雷（开发文档 §6.3）

`core/generation/mine_generator.py`：

```python
"""点后布雷。"""

import random

from ..model.board import Board
from ..model.coordinate import Coordinate


def _safe_zone(click: Coordinate, mode: str, rows: int, cols: int) -> set[Coordinate]:
    """计算禁止布雷区：cell=仅首格，zone3x3=首格 3×3。"""
    if mode == "cell":
        return {click}
    zone: set[Coordinate] = set()
    for dr in (-1, 0, 1):
        for dc in (-1, 0, 1):
            c = Coordinate(click.row + dr, click.col + dc)
            if 0 <= c.row < rows and 0 <= c.col < cols:
                zone.add(c)
    return zone


def place_mines(
    board: Board,
    mine_count: int,
    first_click: Coordinate,
    safe_mode: str,
    rng: random.Random,
) -> None:
    """在安全区之外的格子中随机放置 mine_count 颗雷。"""
    forbidden = _safe_zone(first_click, safe_mode, board.rows, board.cols)
    candidates = [c for c in board.all_coords() if c not in forbidden]
    for c in rng.sample(candidates, mine_count):
        board.cell(c).is_mine = True
```

> 布雷完成后必须调用 §5 的 `calculate_adjacent_mines(board)` 预计算邻域雷数，否则翻开阶段读到的 `adjacent_mines` 全是初始值。

## 5. Core：翻开与洪水填充（开发文档 §6.4）

`core/rules/reveal.py`：

```python
"""翻开与 BFS 洪水填充（邻域雷数在布雷后预计算）。"""

from collections import deque

from ..enums import Mark
from ..model.board import Board
from ..model.coordinate import Coordinate


def count_adjacent_mines(board: Board, coord: Coordinate) -> int:
    """统计 3×3 邻域内的雷数。"""
    return sum(1 for n in board.neighbors8(coord) if board.cell(n).is_mine)


def calculate_adjacent_mines(board: Board) -> None:
    """布雷后调用：遍历棋盘，为每个格子预计算邻域雷数。"""
    for coord in board.all_coords():
        board.cell(coord).adjacent_mines = count_adjacent_mines(board, coord)


def reveal_cell(board: Board, coord: Coordinate) -> bool:
    """翻开单格；返回是否确实翻开（雷/旗/已翻开/越界返回 False）。"""
    if not board.in_bounds(coord):
        return False
    cell = board.cell(coord)
    if cell.is_revealed or cell.is_mine or cell.mark is Mark.FLAG:
        return False
    cell.is_revealed = True
    return True


def flood_fill(board: Board, start: Coordinate) -> list[Coordinate]:
    """从 start 展开空白区（读预计算的 adjacent_mines），返回本次被翻开的全部格子。"""
    revealed: list[Coordinate] = []
    queue: deque[Coordinate] = deque()
    queue.append(start)
    while queue:
        cur = queue.popleft()
        if not reveal_cell(board, cur):
            continue
        revealed.append(cur)
        if board.cell(cur).adjacent_mines == 0:
            queue.extend(board.neighbors8(cur))
    return revealed
```

> `calculate_adjacent_mines` 只在布雷后调用一次（见 §11 首次点击），翻开/连开阶段只读取预计算结果，不再逐格统计。

```python
from ..enums import Mark
from ..model.cell import Cell


def cycle_mark(cell: Cell, marks_enabled: bool) -> int:
    """循环标记；返回剩余雷数的变化量（标旗 -1，去旗 +1，问号 0）。"""
    if cell.is_revealed:
        return 0
    if cell.mark is Mark.NONE:
        cell.mark = Mark.FLAG
        return -1
    if cell.mark is Mark.FLAG:
        cell.mark = Mark.QUESTION if marks_enabled else Mark.NONE
        return 1
    cell.mark = Mark.NONE
    return 0
```

## 7. Core：连开 chord（开发文档 §6.5）

`core/rules/chord.py`：

```python
"""中键/双键连开。"""

from ..enums import Mark
from ..model.board import Board
from ..model.coordinate import Coordinate
from .reveal import flood_fill


def chord(board: Board, coord: Coordinate) -> tuple[list[Coordinate], bool]:
    """连开周围格子；返回 (本次翻开的格子, 是否踩雷)。"""
    cell = board.cell(coord)
    if not cell.is_revealed:
        return [], False
    flags = sum(1 for n in board.neighbors8(coord) if board.cell(n).mark is Mark.FLAG)
    if cell.adjacent_mines != flags:
        return [], False
    opened: list[Coordinate] = []
    for n in board.neighbors8(coord):
        neighbor = board.cell(n)
        if neighbor.mark is Mark.FLAG:
            continue
        if neighbor.is_mine:
            return opened, True  # 踩雷立即中断，不再翻开其余邻居
        opened.extend(flood_fill(board, n))
    return opened, False
```

> 连开踩雷时 `hit=True` 立即返回，Core 不再翻开其余邻居；终局叠加渲染由 UI 依据
> `GameLost(reason=CHORD, coord)` 完成——UI 直接读取 neighbors8(coord) 的预计算状态：未标旗的雷高亮（12）、
> 标对的旗保留旗帜（14）、标错的旗画红叉（11）、其余按已翻开显示数字/空格（开发文档 §4.5）。

```python
"""胜负判定。"""

from ..model.board import Board


def is_won(board: Board, mine_count: int) -> bool:
    """所有非雷格全部翻开即胜（雷是否标旗无关）。"""
    target = board.rows * board.cols - mine_count
    revealed_safe = sum(
        1
        for coord in board.all_coords()
        if board.cell(coord).is_revealed and not board.cell(coord).is_mine
    )
    return revealed_safe == target
```

## 9. Core：Clock（开发文档 §6.7）

`core/clock.py`：

```python
"""计时抽象。Core 不依赖 QTimer。"""

from typing import Protocol


class Clock(Protocol):
    """计时接口。"""

    def start(self) -> None: ...
    def stop(self) -> None: ...
    def pause(self) -> None: ...
    def resume(self) -> None: ...
    def reset(self) -> None: ...

    @property
    def elapsed(self) -> int: ...


class FakeClock:
    """手动推进的时钟，供测试使用。"""

    def __init__(self) -> None:
        self._running = False
        self._paused = False
        self._elapsed = 0

    def start(self) -> None:
        self._running = True

    def stop(self) -> None:
        self._running = False

    def pause(self) -> None:
        self._paused = True

    def resume(self) -> None:
        self._paused = False

    def reset(self) -> None:
        self._running = False
        self._paused = False
        self._elapsed = 0

    def tick(self, seconds: int = 1) -> None:
        """推进 elapsed（仅在运行且未暂停时）。"""
        if self._running and not self._paused:
            self._elapsed = min(999, self._elapsed + seconds)

    @property
    def elapsed(self) -> int:
        return self._elapsed
```

## 11. Core：Game（开发文档 §5.4 / §6.8）

`core/game.py`：

```python
"""Game 聚合根：持有状态、分发命令、发布事件。"""

import random
from dataclasses import dataclass
from typing import Callable

from ..data.difficulty import DifficultyConfig, default_difficulties
from .clock import Clock, FakeClock
from .command import ChangeDifficulty, ChordCell, Command, RestartGame, RevealCell, ToggleMark
from .enums import FaceState, GameStatus, LostReason, Mark
from .event import (
    CellMarked,
    CellRevealed,
    CellsRevealed,
    Event,
    FaceChanged,
    FlagsChanged,
    GameLost,
    GameStateChanged,
    GameWon,
    TimerTicked,
)
from .generation.mine_generator import place_mines
from .model.board import Board
from .model.coordinate import Coordinate
from .rules import chord, marking, reveal, victory


@dataclass
class GameState:
    """游戏状态快照（开发文档 §5.4）。"""

    status: GameStatus
    difficulty: str
    rows: int
    cols: int
    mine_count: int
    remaining_mines: int
    elapsed_seconds: int
    face_state: FaceState


class Game:
    """扫雷逻辑核心，不含任何 Qt。"""

    def __init__(
        self, clock: Clock | None = None, difficulties: DifficultyConfig | None = None
    ) -> None:
        self._clock: Clock = clock or FakeClock()
        self._difficulties: DifficultyConfig = difficulties or default_difficulties()
        self._listeners: list[Callable[[Event], None]] = []
        self._board = Board(9, 9)
        self._rng = random.Random()
        self._first_click: Coordinate | None = None
        self.safe_mode = "zone3x3"
        self.marks_enabled = True
        self.state = GameState(GameStatus.READY, "beginner", 9, 9, 10, 10, 0, FaceState.NORMAL)

    @property
    def board(self) -> Board:
        return self._board

    def subscribe(self, listener: Callable[[Event], None]) -> None:
        """注册事件监听器。"""
        self._listeners.append(listener)

    def _emit(self, event: Event) -> None:
        for listener in self._listeners:
            listener(event)

    def new_game(self, difficulty: str, rows: int, cols: int, mines: int) -> None:
        """开新局：清盘、重置状态，布雷推迟到首次点击。"""
        self._clock.reset()  # 停表并清零（重开从 0 计起）
        self._board = Board(rows, cols)
        self._first_click = None
        self.state = GameState(
            GameStatus.READY, difficulty, rows, cols, mines, mines, 0, FaceState.NORMAL
        )
        self._emit(GameStateChanged(self.state.status))
        self._emit(FlagsChanged(mines))
        self._emit(FaceChanged(FaceState.NORMAL))

    def dispatch(self, command: Command) -> None:
        """分发命令（终局后仅 Restart 有效，见开发文档 §6.8）。"""
        if isinstance(command, RestartGame):
            self.new_game(
                self.state.difficulty, self.state.rows, self.state.cols, self.state.mine_count
            )
            return
        if self.state.status in (GameStatus.WON, GameStatus.LOST):
            return
        if isinstance(command, ChangeDifficulty):
            preset = self._difficulties.preset(command.name)
            self.new_game(command.name, preset.rows, preset.cols, preset.mines)
        elif isinstance(command, RevealCell):
            self._reveal(command.coord)
        elif isinstance(command, ToggleMark):
            self._mark(command.coord)
        elif isinstance(command, ChordCell):
            self._chord(command.coord)

    def tick(self) -> None:
        """每秒被 UI 的 QTimer 调用，发布计时事件。"""
        if self.state.status is GameStatus.PLAYING and self._clock.elapsed >= 999:
            self._clock.stop()
        self._emit(TimerTicked(self._clock.elapsed))

    def _reveal(self, coord: Coordinate) -> None:
        if not self._board.in_bounds(coord):
            return
        if self._first_click is None:
            self._first_click = coord
            place_mines(self._board, self.state.mine_count, coord, self.safe_mode, self._rng)
            reveal.calculate_adjacent_mines(self._board)
            self._clock.start()
            self.state.status = GameStatus.PLAYING
            self._emit(GameStateChanged(GameStatus.PLAYING))
        cell = self._board.cell(coord)
        if cell.is_mine:
            self._end(GameStatus.LOST, LostReason.REVEAL, coord)
            return
        opened = reveal.flood_fill(self._board, coord)
        if opened:
            self._emit(
                CellsRevealed(tuple((c, self._board.cell(c).adjacent_mines) for c in opened))
            )
        if victory.is_won(self._board, self.state.mine_count):
            self._end(GameStatus.WON)

    def _mark(self, coord: Coordinate) -> None:
        if not self._board.in_bounds(coord):
            return
        cell = self._board.cell(coord)
        delta = marking.cycle_mark(cell, self.marks_enabled)
        self.state.remaining_mines += delta
        self._emit(CellMarked(coord, cell.mark))
        self._emit(FlagsChanged(self.state.remaining_mines))

    def _chord(self, coord: Coordinate) -> None:
        if not self._board.in_bounds(coord):
            return
        opened, hit_mine = chord.chord(self._board, coord)
        if opened:
            self._emit(
                CellsRevealed(tuple((c, self._board.cell(c).adjacent_mines) for c in opened))
            )
        if hit_mine:
            self._end(GameStatus.LOST, LostReason.CHORD, coord)
        elif victory.is_won(self._board, self.state.mine_count):
            self._end(GameStatus.WON)

    def _end(
        self,
        status: GameStatus,
        reason: LostReason | None = None,
        coord: Coordinate | None = None,
    ) -> None:
        """终局：停表、切笑脸、发事件；LOST 必须带 reason 与 coord。"""
        self._clock.stop()
        self.state.status = status
        self.state.face_state = FaceState.WINNER if status is GameStatus.WON else FaceState.LOST
        self.state.elapsed_seconds = self._clock.elapsed
        self._emit(FaceChanged(self.state.face_state))
        self._emit(GameStateChanged(status))
        if status is GameStatus.WON:
            self._emit(GameWon())
        else:
            assert reason is not None and coord is not None
            self._emit(GameLost(reason=reason, coord=coord))
```

```python
from PySide6.QtCore import QTimer

from minesweeper_xp.core.clock import Clock


class QtClock(Clock):
    """每 1000ms 递增 elapsed 的时钟。"""

    def __init__(self) -> None:
        self._elapsed = 0
        self._timer = QTimer()
        self._timer.setInterval(1000)
        self._timer.timeout.connect(self._on_timeout)

    def _on_timeout(self) -> None:
        self._elapsed = min(999, self._elapsed + 1)

    def start(self) -> None:
        self._timer.start()

    def stop(self) -> None:
        self._timer.stop()

    def pause(self) -> None:
        self._timer.stop()

    def resume(self) -> None:
        self._timer.start()

    def reset(self) -> None:
        self._timer.stop()
        self._elapsed = 0

    @property
    def elapsed(self) -> int:
        return self._elapsed
```

```python
from PySide6.QtGui import QMouseEvent, QPainter
from PySide6.QtWidgets import QWidget

from minesweeper_xp.core.command import ChordCell, RevealCell, ToggleMark

from minesweeper_xp.core.game import Game
from minesweeper_xp.core.model.coordinate import Coordinate


class BoardWidget(QWidget):
    """把鼠标事件转成 Command 交给 Game，再根据事件重绘。"""

    game_changed = Signal()

    def __init__(self, game: Game) -> None:
        super().__init__()
        self._game = game
        self._game.subscribe(lambda _event: self.game_changed.emit())
        self.game_changed.connect(self.update)
        self.setMouseTracking(True)
        self._tick_timer = QTimer(self)
        self._tick_timer.setInterval(1000)
        self._tick_timer.timeout.connect(self._game.tick)
        self._tick_timer.start()

    def cell_size(self) -> int:
        return 16

    def _coord_at(self, pos) -> Coordinate:
        return Coordinate(pos.y() // self.cell_size(), pos.x() // self.cell_size())

    def mousePressEvent(self, event: QMouseEvent) -> None:
        coord = self._coord_at(event.position())
        if not self._game.board.in_bounds(coord):
            return
        if event.button() == Qt.MouseButton.LeftButton:
            self._game.dispatch(RevealCell(coord))
        elif event.button() == Qt.MouseButton.RightButton:
            self._game.dispatch(ToggleMark(coord))
        elif event.button() == Qt.MouseButton.MiddleButton:
            self._game.dispatch(ChordCell(coord))

    def paintEvent(self, _event) -> None:
        painter = QPainter(self)
        s = self.cell_size()
        for coord in self._game.board.all_coords():
            cell = self._game.board.cell(coord)
            x, y = coord.col * s, coord.row * s
            if cell.is_revealed:
                painter.fillRect(x, y, s, s, Qt.GlobalColor.lightGray)
                if cell.is_mine:
                    painter.fillRect(x, y, s, s, Qt.GlobalColor.black)
                elif cell.adjacent_mines:
                    painter.drawText(
                        x, y, s, s, Qt.AlignmentFlag.AlignCenter, str(cell.adjacent_mines)
                    )
            else:
                painter.fillRect(x, y, s, s, Qt.GlobalColor.gray)
                if cell.mark is Mark.FLAG:
                    painter.drawText(x, y, s, s, Qt.AlignmentFlag.AlignCenter, "F")
                elif cell.mark is Mark.QUESTION:
                    painter.drawText(x, y, s, s, Qt.AlignmentFlag.AlignCenter, "?")
```

#### 14.1.3 `rendering/sprite_loader.py`

```python
"""精灵加载：读位图条、垂直翻转、按图块切分并缓存（开发文档 §8.2）。"""
from pathlib import Path

from PySide6.QtGui import QImage, QPixmap

from .theme import Theme


class SpriteLoader:
    """按主题加载并缓存精灵图块。"""

    def __init__(self, res_dir: Path, theme: Theme) -> None:
        self._res_dir = res_dir
        self._theme = theme
        self._cache: dict[str, list[QPixmap]] = {}

    def _strip(self, sprite_key: str, tile_w: int, tile_h: int) -> list[QPixmap]:
        """加载一张位图条并切成 tile_w×tile_h 图块（带缓存）。"""
        if sprite_key in self._cache:
            return self._cache[sprite_key]
        path = self._res_dir / self._theme.sprites[sprite_key]
        image = QImage(str(path))
        if image.isNull():
            raise FileNotFoundError(f"精灵资源缺失：{path}")
        # BMP 自下而上存储：垂直翻转后图块顺序 == 状态顺序
        image = image.mirrored(False, True)
        pixmap = QPixmap.fromImage(image)
        tiles = [
            pixmap.copy(0, i * tile_h, tile_w, tile_h)
            for i in range(pixmap.height() // tile_h)
        ]
        self._cache[sprite_key] = tiles
        return tiles

    def blocks(self, mono: bool = False) -> list[QPixmap]:
        """16 个方块图块，索引 0–15 对应状态。"""
        return self._strip("blocks_mono" if mono else "blocks", 16, 16)

    def digits(self, mono: bool = False) -> list[QPixmap]:
        """12 个数字图块：0–9、10 空白、11 负号。"""
        return self._strip("digits_mono" if mono else "digits", 13, 23)

    def smiles(self, mono: bool = False) -> list[QPixmap]:
        """5 个笑脸图块：0 正常、1 惊讶、2 失败、3 胜利、4 按下。"""
        return self._strip("smiles_mono" if mono else "smiles", 24, 24)
```

#### 14.1.4 `rendering/block_renderer.py`

```python
"""方块状态 → 图块索引与绘制（开发文档 §4.3 / §8.2）。"""
from PySide6.QtCore import QRect
from PySide6.QtGui import QPainter, QPixmap

from minesweeper_xp.core.enums import Mark
from minesweeper_xp.core.model.cell import Cell


def cell_to_tile(cell: Cell) -> int:
    """Cell 正常状态 → 图块索引 0–15。"""
    if cell.is_revealed:
        return cell.adjacent_mines  # 0–8（雷不会被翻开，走终局覆盖）
    if cell.mark is Mark.FLAG:
        return 14
    if cell.mark is Mark.QUESTION:
        return 13
    return 15


def terminal_tile(cell: Cell) -> int | None:
    """失败终局叠加图块；返回 None 表示按正常状态画（开发文档 §4.5）。"""
    if cell.mark is Mark.FLAG:
        return 11 if not cell.is_mine else None  # 错旗红叉；标对的旗保留旗帜
    if cell.is_mine:
        return 12  # 未标旗的雷：高亮
    return None


def draw_tile(
    painter: QPainter, tiles: list[QPixmap], state: int, x: int, y: int, size: int = 16
) -> None:
    """把图块 state 画到 (x, y)，目标边长 size。"""
    painter.drawPixmap(QRect(x, y, size, size), tiles[state])
```

#### 14.1.5 `rendering/hud_renderer.py`

```python
"""HUD 绘制：LED 数字、笑脸与 3D 边框（开发文档 §8.3 / §8.4）。"""
from PySide6.QtCore import QRect
from PySide6.QtGui import QColor, QPainter, QPixmap

DIGIT_W, DIGIT_H = 13, 23
MINUS = 11  # 负号字形索引


def draw_number(painter: QPainter, digits: list[QPixmap], value: int, x: int, y: int) -> None:
    """画 3 位 LED 数字（value 可为负）。"""
    text = f"{-value:02d}" if value < 0 else f"{value:03d}"
    for i, ch in enumerate(text):
        idx = MINUS if ch == "-" else int(ch)
        painter.drawPixmap(x + i * DIGIT_W, y, digits[idx])


def draw_smile(painter: QPainter, smiles: list[QPixmap], face: int, x: int, y: int) -> None:
    """画 24×24 笑脸。"""
    painter.drawPixmap(x, y, smiles[face])


def draw_bevel(
    painter: QPainter,
    rect: QRect,
    sunken: bool,
    thickness: int = 2,
    base: QColor = QColor("#C0C0C0"),
) -> None:
    """画 3D 凸起/凹陷边框（开发文档 §8.4）。"""
    light = base.lighter(140)
    dark = base.darker(150)
    for i in range(thickness):
        r = rect.adjusted(i, i, -i, -i)
        top_left, bottom_right = (dark, light) if sunken else (light, dark)
        painter.setPen(top_left)
        painter.drawLine(r.left(), r.bottom() - 1, r.left(), r.top())
        painter.drawLine(r.left(), r.top(), r.right() - 1, r.top())
        painter.setPen(bottom_right)
        painter.drawLine(r.right() - 1, r.top(), r.right() - 1, r.bottom() - 1)
        painter.drawLine(r.right() - 1, r.bottom() - 1, r.left(), r.bottom() - 1)
```

```python
    if event.button() == Qt.MouseButton.LeftButton:
        self._pressed = coord
    elif event.button() == Qt.MouseButton.RightButton:
        if self._pressed is not None and self._pressed == coord:
            self._preview = coord  # 左键按住后再按右键：chord 预览
        else:
            self._dispatch(coord, ToggleMark(coord))  # 右键在按下时标记

def mouseReleaseEvent(self, event: QMouseEvent) -> None:
    coord = self._coord_at(event.position())
    if event.button() == Qt.MouseButton.LeftButton:
        if self._preview is not None:
            if coord == self._preview:
                self._dispatch(coord, ChordCell(coord))
            self._preview = None
        elif coord is not None and coord == self._pressed:
            self._dispatch(coord, RevealCell(coord))
        self._pressed = None
    elif event.button() == Qt.MouseButton.RightButton and self._preview is not None:
        self._preview = None
    self.update()
```

#### 14.2.5 终局叠加渲染（开发文档 §4.5）

BoardWidget 订阅事件维护 `_lost`，并在 `_tile_for` 里叠加：

```python
def _on_event(self, event: Event) -> None:
    if isinstance(event, GameLost):
        self._lost = (event.reason, event.coord)
    elif isinstance(event, GameStateChanged) and event.status is not GameStatus.LOST:
        self._lost = None
    self.update()

def _lost_overlay_state(self, coord: Coordinate) -> int | None:
    """失败终局叠加图块；REVEAL 只覆盖触发格，CHORD 覆盖连开点 8 邻居。"""
    if self._lost is None:
        return None
    reason, trigger = self._lost
    cell = self._game.board.cell(coord)
    if reason is LostReason.REVEAL:
        return terminal_tile(cell) if coord == trigger else None
    if coord in self._game.board.neighbors8(trigger):
        return terminal_tile(cell)
    return None
```

规则核对（开发文档 §4.5）：REVEAL 只高亮踩中的雷（12）；CHORD 对 8 邻居叠加——未标旗的雷高亮（12）、标对的旗保留旗帜（`terminal_tile` 返回 None）、错旗红叉（11）、非雷格正常显示。

#### 14.2.6 音效（开发文档 §13）

`audio/sound_manager.py`：

```python
"""音效管理：订阅事件播放（开发文档 §13）。"""
from pathlib import Path

from PySide6.QtCore import QUrl
from PySide6.QtMultimedia import QSoundEffect

from minesweeper_xp.core.event import CellRevealed, Event, GameLost, GameWon, TimerTicked


class SoundManager:
    """把 Core 事件映射为音效。"""

    def __init__(self, res_dir: Path) -> None:
        self._tick = self._load(res_dir / "sounds/tick.wav")
        self._win = self._load(res_dir / "sounds/win.wav")
        self._lose = self._load(res_dir / "sounds/lose.wav")
        self.enabled = True
        self.volume = 0.8
        self.tick_mode = "second"  # second / minute / off

    def _load(self, path: Path) -> QSoundEffect:
        effect = QSoundEffect()
        effect.setSource(QUrl.fromLocalFile(str(path)))
        return effect

    def handle(self, event: Event) -> None:
        """响应 Core 事件播放音效。"""
        if not self.enabled:
            return
        if isinstance(event, GameWon):
            self._win.play()
        elif isinstance(event, GameLost):
            self._lose.play()
        elif isinstance(event, CellRevealed):
            self._tick.play()
        elif isinstance(event, TimerTicked):
            self._play_tick(event.seconds)

    def _play_tick(self, seconds: int) -> None:
        """按 tick_mode 决定是否播滴答声。"""
        if self.tick_mode == "second" or (self.tick_mode == "minute" and seconds % 60 == 0):
            self._tick.play()
```

MainWindow 创建 `SoundManager` 并 `game.subscribe(sound.handle)`；`volume`/`tick_mode` 在阶段 6 接入设置。

#### 14.3.1 `geometry/board_geometry.py`（纯函数，无 Qt）

```python
"""棋盘几何换算（纯函数，开发文档 §9）。"""
from minesweeper_xp.core.model.coordinate import Coordinate

BASE_CELL = 16  # 基准格子边长
BOARD_LEFT = 12  # 棋盘逻辑起点 x
BOARD_TOP = 55  # 棋盘逻辑起点 y
MIN_SCALE = 0.5
MAX_SCALE = 3.0
ZOOM_STEPS = (0.5, 0.75, 1.0, 1.25, 1.5, 2.0, 2.5, 3.0)


def cell_size(scale: float) -> int:
    """按缩放系数返回格子边长。"""
    return round(BASE_CELL * scale)


def window_size(rows: int, cols: int, scale: float) -> tuple[int, int]:
    """按缩放系数返回窗口逻辑尺寸（宽、高）。"""
    return round((cols * BASE_CELL + 24) * scale), round((rows * BASE_CELL + 65) * scale)


def fit_scale(rows: int, cols: int, avail_w: int, avail_h: int) -> float:
    """由可用尺寸反推自适应缩放系数（夹在 MIN/MAX 之间）。"""
    s = min(avail_w / (cols * BASE_CELL + 24), avail_h / (rows * BASE_CELL + 65))
    return max(MIN_SCALE, min(MAX_SCALE, s))


def widget_pos_to_cell(
    px: float, py: float, scale: float, rows: int, cols: int
) -> Coordinate | None:
    """窗口逻辑坐标 → 棋盘坐标；越界返回 None。"""
    col = int((px / scale - BOARD_LEFT) / BASE_CELL)
    row = int((py / scale - BOARD_TOP) / BASE_CELL)
    if 0 <= row < rows and 0 <= col < cols:
        return Coordinate(row, col)
    return None
```

#### 14.3.2 BoardWidget 接入缩放

```python
self._scale: float = 1.0

def resizeEvent(self, event) -> None:
    super().resizeEvent(event)
    board = self._game.board
    new_scale = fit_scale(board.rows, board.cols, self.width(), self.height())
    if abs(new_scale - self._scale) > 0.01:
        self._scale = new_scale
        self.update()

def wheelEvent(self, event) -> None:
    if event.modifiers() & Qt.KeyboardModifier.ControlModifier:
        idx = ZOOM_STEPS.index(self._scale) if self._scale in ZOOM_STEPS else 2
        idx += 1 if event.angleDelta().y() > 0 else -1
        self._scale = ZOOM_STEPS[max(0, min(len(ZOOM_STEPS) - 1, idx))]
        self.resize(*window_size(self._game.board.rows, self._game.board.cols, self._scale))
        self.update()
        event.accept()
```

#### 14.3.4 DPI 说明

- Qt 6 默认启用高 DPI，QPainter 绘制用逻辑像素，无需手动处理 devicePixelRatio。

- 在系统缩放 100/125/150/200% 下逐档目测：无模糊、无错位、无裁剪。
